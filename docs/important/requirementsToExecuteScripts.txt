function Invoke-RepoScript {
    param(
        [Parameter(Mandatory = $true)]
        [string]$RelativeScriptPath,

        [Parameter(ValueFromRemainingArguments = $true)]
        [object[]]$ScriptArgs
    )

    $dir = (Get-Location).Path

    while ($true) {
        $scriptPath = Join-Path $dir $RelativeScriptPath

        if (Test-Path $scriptPath) {
            & $scriptPath @ScriptArgs
            return
        }

        $parent = Split-Path -Path $dir -Parent

        if ([string]::IsNullOrWhiteSpace($parent) -or $parent -eq $dir) {
            break
        }

        $dir = $parent
    }

    Write-Error "No se encontró $RelativeScriptPath en esta carpeta ni en carpetas padre."
}

function tr {
    param(
        [switch]$f
    )

    $dir = (Get-Location).Path

    while ($true) {
        $scriptPath = Join-Path $dir "scripts/tr.ps1"

        if (Test-Path $scriptPath) {
            $params = @{}
            if ($f.IsPresent) { $params["f"] = $true }
            & $scriptPath @params
            return
        }

        $parent = Split-Path -Path $dir -Parent
        if ([string]::IsNullOrWhiteSpace($parent) -or $parent -eq $dir) { break }
        $dir = $parent
    }

    Write-Error "No se encontró scripts/tr.ps1 en esta carpeta ni en carpetas padre."
}

function crawler {
    param(
        [switch]$s,
        [switch]$d
    )

    $dir = (Get-Location).Path

    while ($true) {
        $scriptPath = Join-Path $dir "scripts/run-crawler.ps1"

        if (Test-Path $scriptPath) {
            $params = @{}
            if ($d.IsPresent) { $params["d"] = $true }
            if ($s.IsPresent) { $params["s"] = $true }
            & $scriptPath @params
            return
        }

        $parent = Split-Path -Path $dir -Parent
        if ([string]::IsNullOrWhiteSpace($parent) -or $parent -eq $dir) { break }
        $dir = $parent
    }

    Write-Error "No se encontró scripts/run-crawler.ps1 en esta carpeta ni en carpetas padre."
}
