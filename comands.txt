--------------------------------------------------------------

    cd .\backend\

    .venv\Scripts\activate
    pip install -r requirements.txt
    playwright install chromium

--------------------------------------------------------------
    git pull
    cd .\backend\
    .venv\Scripts\activate
    pip install -r requirements.txt

        pip-review --auto
